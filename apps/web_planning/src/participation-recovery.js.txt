"use strict";
const grant=document.querySelector('meta[name="participation"]').content;
const status=document.getElementById('recovery-status'),button=document.getElementById('recover'),choice=document.getElementById('pending-choice');
(async()=>{
  try{
    const rows=await participantJournal(grant,'get');
    status.textContent=rows.length?'Your pending returns are available locally. Their saved status is unknown.':'No pending return was found in this browser. Try the original browser profile or local client journal.';
    if(!rows.length)return;
    for(const row of rows)choice.append(new Option(row.message+' · '+row.request_key,row.request_key));
    choice.value=rows.some(row=>row.request_key===participantPreferred(grant))?participantPreferred(grant):rows[0].request_key;
    function selected(){const body=rows.find(row=>row.request_key===choice.value);participantPreferred(grant,body.request_key);document.getElementById('pending-summary').textContent=body.message+'\nNext: '+body.next_action+'\nRequest key: '+body.request_key;return body;}
    choice.parentElement.hidden=false;choice.onchange=selected;selected();
    button.hidden=false;button.onclick=()=>{
      const value={format:'augnes_participant_pending.v1',grant_id:grant,origin:location.origin,body:selected()};
      const url=URL.createObjectURL(new Blob([JSON.stringify(value,null,2)],{type:'application/json'})),a=document.createElement('a');a.href=url;a.download='augnes-pending-return.json';a.click();setTimeout(()=>URL.revokeObjectURL(url),1000);
    };
  }catch{status.textContent='Local recovery storage is unavailable. Use a previously downloaded recovery copy or local client journal.';}
})();
