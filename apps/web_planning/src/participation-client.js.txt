async function request(path,body){
  const response=await fetch(root+path,{method:body===undefined?'GET':'POST',redirect:'manual',cache:'no-store',credentials:'same-origin',
    headers:body===undefined?{}:{'Content-Type':'application/json','X-CSRF-Token':csrf},...(body===undefined?{}:{body:JSON.stringify({...body,workspace_id:state.workspace_id,project_id:state.project_id})})});
  if(response.headers.get('x-web-planning-response')!=='1'){loseAccess();throw Error('participant_login_required');}
  if(!response.ok){const error=(await response.json().catch(()=>({}))).error??'request_unavailable';
    if(response.status===403||response.status===410)loseAccess();
    throw Error(error);}
  return response;
}
const json=async(path,body)=>(await request(path,body)).json();
async function digest(bytes){return 'sha256:'+Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256',bytes)),b=>b.toString(16).padStart(2,'0')).join('');}
function download(bytes,name){const url=URL.createObjectURL(new Blob([bytes],{type:'application/octet-stream'})),a=document.createElement('a');a.href=url;a.download=name;a.click();setTimeout(()=>URL.revokeObjectURL(url),1000);}
async function bytes(file){
  if(!file.download.startsWith(root+'/files/'))throw Error('scope_denied');
  const data=new Uint8Array(await (await request(file.download.slice(root.length))).arrayBuffer());
  if(data.length!==file.bytes||await digest(data)!==file.digest)throw Error('file_content_integrity');return data;
}
function drawFile(parent,file,original){
  const section=document.createElement('article');text(section,file.name+' · '+file.bytes+' bytes · '+file.digest);
  button(section,'Download '+file.name,async()=>download(await bytes(file),file.name));
  if(/\.(txt|md|markdown|log|json|patch|mjs)$/i.test(file.name)){
    const reader=document.createElement('div');section.append(reader);
    let next=0;const read=button(reader,'Read '+file.name,async()=>{
      const page=await json(file.download.slice(root.length)+'/text',{offset:next,find:''});
      if(page.file.digest!==file.digest||page.total!==file.bytes)throw Error('file_content_integrity');
      const pre=document.createElement('pre');pre.textContent=page.text;reader.append(pre);
      text(reader,'Bytes '+page.start+'–'+page.end+' of '+page.total+(page.next===null?' · End of file':' · Partial view'));
      next=page.next;read.disabled=next===null;read.textContent='Continue reading '+file.name;
    });
    if(original&&file.bytes<=32768&&/\.(txt|md|markdown)$/i.test(file.name))button(section,'Edit '+file.name,async()=>{
      if(pendingChoices.length||!state.can_submit)throw Error('Check the pending or answered request first');
      const m=await import(root+'/exchange-text.mjs'),data=await bytes(file);
      edit={file,source:m.decodeExchangeText(data,file.name),replace:m.replaceExchangePassage};
      $('edit').hidden=false;$('edit-name').textContent=file.name;$('upload').disabled=$('review').disabled=true;$('find').focus();
    });
  }
  parent.append(section);
}

async function send(resolve){
  const body=pending;if(!body)return;
  if(!resolve)await journal('put',body);
  const {operation,...input}=body;
  const route=operation==='retain'?(resolve?'/retain-resolve':'/retain'):operation==='project-return'?(resolve?'/project-return-resolve':'/project-return'):(resolve?'/resolve':'/save');
  const result=await json(route,input);
  $('retry').hidden=true;$('release').hidden=true;
  if(result.outcome==='saved'){
    choosePending(await journal('delete',body),null);$('return').reset();edit=null;$('edit').hidden=true;await refresh();if(operation){await loadOwnWorks();}status(operation==='retain'?'Your work is saved. Open it to prepare your contribution; return selected saved results here.':'Return saved. The owner can now inspect and decide.');
  }else{
    $('retry').hidden=!result.retry_allowed;$('release').hidden=result.retry_allowed;
    status(result.retry_allowed?'No matching save observed. Only retry this exact pending return; a request may still be in flight.':'This return was not recorded and its save basis is no longer eligible. Keep its recovery copy before starting again.');
  }
}

    selected=[new File([edit.replace(edit.source,find,replacement)],edit.file.name,{type:edit.file.media_type??''})];
