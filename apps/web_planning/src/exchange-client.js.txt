// These controls use the active private Web principal and the selected task.
let exchangeDraft = null, exchangePending = null, exchangeState = null;
const exchangeStoragePrefix = 'augnes.pending-file.' + scope.workspace_id + '.' + scope.project_id + '.';
function forgetExchange(id) {
  try { sessionStorage.removeItem(exchangeStoragePrefix + id); } catch {}
}
function forgetPrivateExchanges() {
  try { for (const key of Object.keys(sessionStorage)) if (key.startsWith(exchangeStoragePrefix)) sessionStorage.removeItem(key); } catch {}
}
function clearExchange() {
  exchangeDraft = exchangePending = exchangeState = null;
  $('exchange-form')?.reset();
  $('exchange-list')?.replaceChildren();
}
function exchangeControls() {
  $('exchange-form').hidden = !exchangeDraft;
  const edit = exchangeDraft?.textEdit;
  $('exchange-file-fields').hidden = !!edit;
  $('exchange-upload').disabled = !!edit;
  $('exchange-review-field').hidden=!exchangeDraft?.basis || !!edit;
  $('exchange-upload').required=!edit && (!exchangeDraft?.basis || !$('exchange-review').value);

  $('exchange-text-fields').hidden = !edit;
  $('exchange-text-fields').disabled = !edit;
  $('exchange-edit-name').textContent = edit ? edit.file.name + ' · version ' + exchangeDraft.basis.revision : '';
  $('exchange-fields').disabled = !!exchangePending;
  $('exchange-submit').disabled = !!exchangeDraft?.answered;
  $('exchange-new').hidden = !!exchangeDraft;
  $('exchange-uncertain').hidden = !exchangePending;
  $('exchange-retry').hidden = true;
}
function beginExchange(source, textEdit = null) {
  // A reconciled, unrecorded return can be re-entered from the same original
  // without storing its private text in tab storage or requiring a local file.
  if (textEdit && exchangeDraft?.basis?.fingerprint === source.ref.fingerprint &&
      !busy && !exchangePending && !exchangeDraft.answered && !exchangeDraft.textEdit && !$('exchange-upload').files.length) {
    exchangeDraft.textEdit = textEdit;
    exchangeControls(); $('exchange-find').focus(); return;
  }
  if (busy || pending || dirty || directionDirty || relationPending || sceneReviewPending || exchangeDraft) {
    status('Finish the current draft before starting a file request or return.'); return;
  }
  $('edit-fields').disabled = true;
  exchangeDraft = { basis: source?.ref ?? null, expected: exchangeState.head, textEdit };
  $('exchange-form').reset();
  $('exchange-legend').textContent = source ? 'Return files for request · version ' + source.ref.revision : 'New file request';
  $('exchange-message-label').textContent = source ? 'What did you change or learn?' : 'What would you like done?';
  $('exchange-submit').textContent = source ? 'Send returned files' : 'Send request';
  $('exchange-next').required = !!source;
  exchangeControls();
  $(textEdit ? 'exchange-find' : 'exchange-message').focus();
}
async function readExchangeBytes(file) {
  const response = await request(file.download);
  const bytes = new Uint8Array(await readPrivate(response,'arrayBuffer'));
  const digest = await exchangeDigest(bytes);
  if (bytes.length !== file.bytes || digest !== file.digest) {
    status('Downloaded bytes do not match this version. Reopen the task before trying again.'); return;
  }
  if (!accessLost) return bytes;
}
async function downloadExchangeFile(file) {
  const bytes = await readExchangeBytes(file);
  if (!bytes) return;
  const url = URL.createObjectURL(new Blob([bytes],{type:'application/octet-stream'}));
  const link = document.createElement('a'); link.href = url; link.download = file.name; link.click();
  setTimeout(() => URL.revokeObjectURL(url),1000);
}
function addTextReader(container, file, source) {
  if (!textFile(file.name) && !/\.log$/i.test(file.name)) return;
  if (file.bytes > 32768 || !textFile(file.name)) { addPagedReader(container,file); return; }
  const read = document.createElement('button');
  read.type = 'button'; read.textContent = 'Read text';
  read.setAttribute('aria-label', 'Read ' + file.name + ' as text');
  container.append(read);
  read.onclick = () => run(async () => {
    const bytes = await readExchangeBytes(file);
    if (!bytes) return;
    const { decodeExchangeText, replaceExchangePassage } = await import('/exchange-text.mjs');
    if (accessLost || !container.isConnected) return;
    let sourceText;
    try { sourceText = decodeExchangeText(bytes, file.name); }
    catch { status('This file cannot be displayed as UTF-8 text. Use its download link.'); return; }
    const text = document.createElement('pre');
    text.textContent = sourceText;
    read.replaceWith(text);
    if (source && sourceText.length) {
      const edit = document.createElement('button');
      edit.type = 'button'; edit.textContent = 'Edit this text';
      edit.onclick = () => beginExchange(source, { file, sourceText, replaceExchangePassage });
      container.append(edit);
    }
  });
}
// Each page is a fresh authorized read of the exact listed file; no private
// text is cached in tab storage. Byte positions include BOMs and line endings.
function addPagedReader(container,file) {
  const button=(label)=>{const el=document.createElement('button');el.type='button';el.textContent=label;return el;};
  const open=button('Read text');open.setAttribute('aria-label','Read '+file.name+' as text');container.append(open);
  open.onclick=()=>{
    const panel=document.createElement('section'),position=document.createElement('input'),find=document.createElement('input');
    position.type='number';position.min='0';position.max=String(file.bytes);position.value='0';
    find.type='search';find.placeholder='Exact text to find';
    const label=(name,input)=>{const l=document.createElement('label');input.id='read-'+file.download.split('/').slice(-3).join('-')+'-'+input.type;l.htmlFor=input.id;l.textContent=name;l.append(input);return l;};
    const read=button('Read from position'),search=button('Find from position'),next=button('Continue reading');
    const info=document.createElement('p'),text=document.createElement('pre');
    info.setAttribute('aria-live','polite');next.disabled=true;
    panel.append(label('Start byte',position),label('Find in this version',find),read,search,next,info,text);open.replaceWith(panel);
    let continuation=null;
    const page=(at,needle)=>run(async()=>{
      const result=await api(file.download+'/text',{offset:at,find:needle});
      if(accessLost || !panel.isConnected)return;
      if(result.file.digest!==file.digest || result.file.name!==file.name || result.total!==file.bytes ||
          !file.download.includes('/'+result.ref.work_id+'/exchange-files/'+result.ref.revision+'/'+result.ref.fingerprint.slice(7)+'/'))throw Error('file_revision_unavailable');
      position.value=String(result.start);continuation=result.next;next.disabled=continuation===null;
      const extent='Bytes '+result.start+'–'+result.end+' (end exclusive) of '+result.total+' · version '+result.ref.revision+' · '+file.digest;
      info.textContent=result.found===false?'No match from byte '+at+' to the end. '+extent:
        extent+(result.next!==null?' · Partial view. More text follows.':result.start?' · End of file. This is a partial view.':' · Complete text.');
      text.textContent=result.text;
    });
    read.onclick=()=>page(Number(position.value),'');search.onclick=()=>{if(find.value)page(Number(position.value),find.value);};
    next.onclick=()=>{if(continuation!==null)page(continuation,'');};page(0,'');
  };
}
function textReturnFile(edit) {
  let find = $('exchange-find').value, replacement = $('exchange-replacement').value;
  // A textarea represents every line separator as LF. Preserve a document's
  // separator, and don't guess when a multiline passage could be ambiguous.
  if (find.includes('\n') || replacement.includes('\n')) {
    const separators = new Set(edit.sourceText.match(/\r\n|\r|\n/g));
    if (separators.size > 1) throw new Error('Choose a single-line edit for a file with mixed line endings.');
    const [separator = '\n'] = separators;
    find = find.replaceAll('\n', separator);
    replacement = replacement.replaceAll('\n', separator);
  }
  const bytes = edit.replaceExchangePassage(edit.sourceText, find, replacement);
  return new File([bytes], edit.file.name);
}
async function exchangeDigest(bytes) {
  const sum = new Uint8Array(await crypto.subtle.digest('SHA-256',bytes));
  return 'sha256:' + [...sum].map(b=>b.toString(16).padStart(2,'0')).join('');
}
function drawExchanges(state) {
  exchangeState = state;
  const list = $('exchange-list');
  list.replaceChildren();
  for (const item of state.items) {
    if (item.kind !== 'request') continue;
    const reply = state.items.find(r => r.basis?.fingerprint === item.ref.fingerprint);
    const article = document.createElement('article');
    const heading = document.createElement('h3');
    heading.textContent = (reply ? 'Answered request' : 'Awaiting a return') + ' · version ' + item.ref.revision;
    article.append(heading);
    for (const part of [item, reply].filter(Boolean)) {
      const message = document.createElement('p');
      message.textContent = (part.kind === 'request' ? 'Request: ' : 'Return: ') + part.message;
      message.id = 'exchange-version-' + part.ref.revision;
      message.tabIndex = -1;
      article.append(message);
      const files = document.createElement('ul');
      for (const file of part.files) {
        const li = document.createElement('li'), link = document.createElement('a');
        link.href = file.download; link.download = file.name;
        link.onclick = event => { event.preventDefault(); run(() => downloadExchangeFile(file)); };
        link.textContent = file.name + ' · version ' + part.ref.revision + ' · ' + file.bytes + ' bytes';
        li.append(link); files.append(li);
        addTextReader(li, file, !reply && part === item && item.files.length === 1 ? item : null);
      }
      article.append(files);
      if (part.next_action) {
        const next = document.createElement('p'); next.textContent = 'Next action: ' + part.next_action; article.append(next);
      }
    }
    if (!reply) {
      const button = document.createElement('button'); button.type = 'button';
      button.textContent = 'Return edited files'; button.onclick = () => beginExchange(item); article.append(button);
    }
    list.append(article);
  }
  if (!list.children.length) list.textContent = 'No file requests yet.';
}
async function readExchanges() {
  const target = work;
  const data = await api('/api/work/' + target + '/exchanges');
  if (target !== work || accessLost) return;
  drawExchanges(data);
  $('exchange-new').disabled = false;
  $('exchange-refresh').hidden = true;
}
function openExchanges(revision, view) {
  const previous = exchangeState;
  clearExchange();
  $('exchange-panel').hidden = !revision;
  exchangeControls();
  if (!revision) return;
  if (!view && revision.revision === 1) view = { head: binding(revision), items: [] };
  if (!view && previous?.head.fingerprint === revision.predecessor && !revision.exchange)
    view = { head: binding(revision), items: previous.items };
  $('exchange-new').disabled = !view;
  $('exchange-refresh').hidden = !!view;
  if (view) {
    drawExchanges(view);
    let marker;
    try { marker = JSON.parse(sessionStorage.getItem(exchangeStoragePrefix + revision.work_id)); } catch {}
    if (marker) {
      const observed = view.items.find(item => item.request_key === marker.request_key);
      if (observed && !sameExchangeMarker(observed,marker)) {
        const conflict=document.createElement('p');conflict.textContent='This request key was used for different material. The earlier draft is retained; review this task before starting another request.';
        $('exchange-list').prepend(conflict);exchangeDraft={basis:marker.basis,expected:marker.expected,answered:true};
        $('exchange-message').value=marker.message;$('exchange-next').value=marker.next_action;$('edit-fields').disabled=true;exchangeControls();return;
      }
      if (observed) {
        forgetExchange(revision.work_id);
        const outcome = document.createElement('p');
        outcome.textContent = 'Previous file save confirmed at version ' + observed.ref.revision + '.';
        $('exchange-list').prepend(outcome);
      } else {
        exchangeDraft = { basis:marker.basis, expected:marker.expected };
        const advanced = view.head.fingerprint !== marker.expected.fingerprint;
        exchangePending = advanced ? null : { id:revision.work_id, marker };
        $('edit-fields').disabled = true;
        $('exchange-legend').textContent = marker.basis ? 'Recover file return' : 'Recover file request';
        $('exchange-message').value = marker.message; $('exchange-next').value = marker.next_action;
        $('exchange-next').required = !!marker.basis;
        $('exchange-message-label').textContent = marker.basis ? 'What did you change or learn?' : 'What would you like done?';
        $('exchange-submit').textContent = marker.basis ? 'Send returned files' : 'Send request';
        exchangeControls();
        if (advanced) {
          $('exchange-refresh').hidden = false;
          const guidance = document.createElement('p');
          guidance.textContent = 'Your earlier save was not recorded. Your draft is retained. Read latest file requests, then reselect: ' + marker.files.map(f => f.name).join(', ') + (marker.basis ? '. For an in-page edit, reopen the original text.' : '');
          $('exchange-list').prepend(guidance);
        }
      }
    }
  }
  else $('exchange-list').textContent = 'Read the latest requests and returns for this saved version.';
}
function sameExchangeMarker(item,marker) {
  const sameRef=(a,b)=>a===null || b===null?a===b:a.work_id===b.work_id && a.revision===b.revision && a.fingerprint===b.fingerprint;
  return item.expected?.revision===marker.expected.revision && item.expected.fingerprint===marker.expected.fingerprint &&
    sameRef(item.basis,marker.basis) && item.kind===marker.kind && item.message===marker.message.trim() && item.next_action===marker.next_action.trim() &&
    item.files.length===marker.files.length && item.files.every((f,i)=>['name','bytes','digest','media_type'].every(k=>f[k]===marker.files[i][k]));
}
async function sendExchange(resolve) {
  const retained = exchangePending;
  try {
    if (retained.marker) {
      const view = await api('/api/work/' + retained.id + '/exchanges');
      const match = view.items.find(item => item.request_key === retained.marker.request_key);
      if (!match && view.head.revision === retained.marker.expected.revision) {
        exchangePending = null; exchangeDraft.recover = retained.marker; exchangeControls();
        status('This request is not saved. Reselect the files: ' + retained.marker.files.map(f=>f.name).join(', ') + (retained.marker.basis ? '. For an in-page edit, reopen the original text.' : '')); return;
      }
      const latest = await api('/api/work/' + retained.id);
      await show(latest.saved,latest.exchanges); return;
    }
    const response = await api('/api/work/' + retained.id + '/exchanges/' + (resolve ? 'resolve' : 'save'), retained.body);
    if (response.outcome !== 'saved') {
      $('exchange-retry').hidden = false; status('No matching save was found. You may retry the exact retained files.'); return;
    }
    forgetExchange(retained.id); clearExchange(); exchangeControls();
    const latest = await api('/api/work/' + retained.id);
    await show(latest.saved, latest.exchanges); status('Files saved at version ' + response.item.ref.revision + '.');
  } catch (error) {
    if (accessLost) return;
    if (!exchangePending) {
      $('edit-fields').disabled = false;
      $('exchange-refresh').hidden = false;
      status('Files were saved. Reopen the task to read them.'); return;
    }
    if (error.status && error.status < 500 && !resolve) {
      forgetExchange(retained.id);
      exchangePending = null; exchangeControls();
      $('exchange-refresh').hidden = false;
      status((messages[error.message] || error.message.replaceAll('_', ' ')) + '. Your input is retained. Read latest file requests before sending again.');
    } else {
      exchangeControls(); status('Save outcome unknown. Check file save before another write.');
    }
  }
}

$('exchange-refresh').onclick = () => run(async () => {
  await readExchanges();
  if (exchangeDraft) {
    exchangeDraft.expected = exchangeState.head;
    exchangeDraft.answered = !!exchangeDraft.basis && exchangeState.items.some(r => r.basis?.fingerprint === exchangeDraft.basis.fingerprint);
    exchangeControls();
    if (exchangeDraft.answered) {
      status('This request already has a return. Your draft is available to copy; cancel it to select another request.'); return;
    }
  }
  status('Latest requests and returns are shown. Review them before sending the retained draft.');
});
$('exchange-form').onsubmit = event => {
  event.preventDefault();
  run(async () => {
    if (!exchangeDraft || exchangeDraft.answered || exchangePending || !$('exchange-refresh').hidden) return;
    let selected;
    const edit = exchangeDraft.textEdit;
    if (edit) {
      try { selected = [textReturnFile(edit)]; }
      catch (error) { status(error.message); return; }
    } else {
      selected = [...$('exchange-upload').files];
      if(exchangeDraft.basis && $('exchange-review').value){
        const review=new File([$('exchange-review').value],'review.md',{type:'text/markdown'});
        if(review.size>32768){status('Keep the written review within 32 KiB, or attach a file.');return;}
        selected.push(review);
      }
    }
    if (!selected.length || selected.length > 8 || selected.some(f => f.size > 6291456) || selected.reduce((n,f) => n + f.size, 0) > 7340032) {
      status('Choose 1–8 files within the displayed size limits. Your draft is retained.'); return;
    }
    const {basis,expected,recover} = exchangeDraft;
    const message = $('exchange-message').value, next_action = $('exchange-next').value;
    const files = [];
    for (const f of selected) {
      const bytes = new Uint8Array(await f.arrayBuffer());
      const pieces=[];
      for(let i=0;i<bytes.length;i+=24576)pieces.push(btoa(String.fromCharCode(...bytes.subarray(i,i+24576))));
      const digest = await exchangeDigest(bytes);
      files.push({ name: f.name, media_type: edit ? edit.file.media_type : f.type || null, bytes: f.size, digest, data: pieces.join('') });
    }
    const request_key = recover?.request_key ?? crypto.randomUUID();
    const body = { basis, expected, kind:basis ? 'return' : 'request', request_key, message, next_action, files };
    const marker = {...body,files:files.map(({data,...identity})=>identity)};
    if (recover && JSON.stringify(recover)!==JSON.stringify(marker)) {
      status('The selected files or message differ from the pending request. Restore the original selection or cancel after reviewing this task.'); return;
    }
    try { sessionStorage.setItem(exchangeStoragePrefix + work,JSON.stringify(marker)); } catch {
      status('The browser could not retain the save for recovery. Enable tab storage or use the local file client. Your draft is retained.'); return;
    }
    exchangePending = { id:work, body };
    exchangeControls();
    await sendExchange(false);
  });
};
