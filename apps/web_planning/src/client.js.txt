async function request(path,body){
  const response=await fetch(path,{method:body===undefined?'GET':'POST',credentials:'same-origin',cache:'no-store',redirect:'error',
    headers:body===undefined?{}:{'Content-Type':'application/json','X-CSRF-Token':csrf},body:body===undefined?undefined:JSON.stringify({...scope,...body}),signal:AbortSignal.timeout(15000)});
  if(response.status===403||response.status===401){deny();throw new Error('access_denied');}
  if(!response.ok){let code='storage_unavailable';try{code=(await response.json()).error;}catch{};const e=new Error(code);e.status=response.status;throw e;}
  return response;
}

function draft(){const notes=[...$('notes').children].map(row=>Object.fromEntries([...row.querySelectorAll('[data-field]')].map(e=>[e.dataset.field,e.dataset.field==='observed_at'?(e.value||null):e.value])));

function changed(){dirty=true;$('saved-label').textContent=saved?`Unsaved edits · based on revision ${saved.revision}`:'Unsaved draft';reviewChanges();}
function reviewChanges(){const d=draft();const lines=[];
  if(!saved)lines.push('New work will be created only when you press Save.');
  else {for(const [k,label] of [['goal','Goal'],['success_criteria','Success criteria'],['non_goals','Non-goals']])
    if(JSON.stringify(d.definition[k])!==JSON.stringify(saved.definition[k]))lines.push(label+' edited.');
    const before=saved.sources.map(noteInput),after=d.notes;
    if(JSON.stringify(before)!==JSON.stringify(after))lines.push('Selected context edited. Originals remain in earlier revisions.');
    if(!lines.length)lines.push('No definition or selection edits. Save records another deliberate revision.');}
  $('change-list').textContent=lines.join(' ');
}
