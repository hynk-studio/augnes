async function request(path,body){
  if(accessLost)throw new Error('access_denied');
  const response=await fetch(path,{method:body===undefined?'GET':'POST',credentials:'same-origin',cache:'no-store',redirect:'manual',
    headers:{'X-Requested-With':'XMLHttpRequest',...(body===undefined?{}:{'Content-Type':'application/json','X-CSRF-Token':csrf})},body:body===undefined?undefined:JSON.stringify({...scope,...body}),signal:AbortSignal.timeout(15000)});
  // Access can return 401/403, a login redirect, or an HTML challenge. None is
  // product data. An ordinary storage/transport failure still retains the draft.
  if(accessLost||response.status===403||response.status===401||response.type==='opaqueredirect'||
    (response.ok&&response.headers.get('X-Web-Planning-Response')!=='1')){deny();throw new Error('access_denied');}
  if(!response.ok){let code='storage_unavailable';try{code=(await response.json()).error;}catch{};const e=new Error(code);e.status=response.status;throw e;}
  return response;
}

function validCapacity(data,check,expected){
  const q=value=>value && Number.isSafeInteger(value.limit) && value.limit>0 && (value.used===null?value.remaining===null:Number.isSafeInteger(value.used)&&value.used>=0&&value.remaining===value.limit-value.used);
  return data?.format==='web_planning_capacity.v0.1' && data.check_id===check && JSON.stringify(data.expected)===JSON.stringify(expected) &&
    typeof data.fits==='boolean' && typeof data.noop==='boolean' && Array.isArray(data.issues) && data.issues.every(i=>typeof i.code==='string' && /^[a-z_]+$/.test(i.code) && (i.note===null||Number.isSafeInteger(i.note))) &&
    ['notes','selection','definition','relations','history'].every(key=>q(data[key])) && Array.isArray(data.rows) && data.rows.length===(data.notes.used!==null&&data.notes.used<=8?data.notes.used:0) &&
    data.rows.every(r=>['characters','source_units','bytes','text_bytes','duplicate_of'].every(k=>r[k]===null||(Number.isSafeInteger(r[k])&&r[k]>=0))) &&
    Number.isSafeInteger(data.note_character_limit) && Number.isSafeInteger(data.source_unit_limit) && Number.isSafeInteger(data.request_byte_limit) &&
    (data.selection.used===null || (Number.isSafeInteger(data.selection.text_bytes)&&data.selection.text_bytes>=0&&data.selection.metadata_bytes>=0&&data.selection.text_bytes+data.selection.metadata_bytes===data.selection.used)) &&
    (!data.fits || (!data.issues.length && ['notes','selection','definition','relations','history'].every(k=>data[k].used!==null&&data[k].remaining>=0) && data.rows.every(r=>r.characters!==null&&r.bytes!==null&&r.text_bytes!==null&&r.source_units!==null) && (data.history.remaining>0||data.noop)));
}

  try {

  finally{capacityInFlight=false;if(capacityQueued&&!accessLost)queueCapacity(0);}
}

function renderFiles(){
  $('file-selection').replaceChildren();
  selectedFiles.forEach((file,i)=>{
    const row=document.createElement('section'),p=document.createElement('p');
    p.textContent=`${file.name} · ${fileSize(file).toLocaleString('en-US')} bytes · ${file.data===undefined?'retained saved identity':'new selection, not uploaded'}`;
    const label=document.createElement('label');label.textContent=`Role for ${file.name}`;
    const select=document.createElement('select');select.id='file-role-'+i;label.htmlFor=select.id;select.setAttribute('aria-label',label.textContent);
    for(const role of ['report','source','results','other']){const option=document.createElement('option');option.value=role;option.textContent=role;select.append(option);}select.value=file.role;
    // A retained descriptor includes its authored role. Reselect bytes to change that attribution.
    select.disabled=file.data===undefined;select.onchange=()=>{file.role=select.value;changed();};label.append(select);
    const remove=document.createElement('button');remove.type='button';remove.textContent=`Remove ${file.name} from next revision`;
    remove.onclick=()=>{selectedFiles.splice(i,1);fileSelectionVersion++;renderFiles();changed();};
    row.append(p,label,remove);$('file-selection').append(row);
  });
  const bytes=selectedFiles.reduce((n,f)=>n+fileSize(f),0);
  $('file-capacity').textContent=`${selectedFiles.length} / 8 files; ${bytes.toLocaleString('en-US')} / 524,288 selected bytes. ${bytes>524288?(bytes-524288).toLocaleString('en-US')+' bytes over; revise this selection.':'History capacity and final admission are checked on Save.'}`;
}
function fileSize(file){return file.bytes??(file.data.length/4*3-(file.data.endsWith('==')?2:file.data.endsWith('=')?1:0));}

  const picked=[...$('work-files').files],version=fileSelectionVersion;
  if(picked.length+selectedFiles.length>8){status(messages.file_count_exceeded+' Existing draft files are retained.');return;}
  if(picked.some(f=>f.size>262144)){status(messages.file_bytes_exceeded+' Existing draft files are retained.');return;}
  if(new Set([...selectedFiles,...picked].map(f=>f.name)).size!==selectedFiles.length+picked.length){status('File names must be unique. Remove the prior selection explicitly before replacing it; earlier revisions keep its bytes.');return;}
  const added=[];
  for(const file of picked){const bytes=new Uint8Array(await file.arrayBuffer());let binary='';for(let i=0;i<bytes.length;i+=8192)binary+=String.fromCharCode(...bytes.subarray(i,i+8192));added.push({name:file.name,role:'other',data:btoa(binary)});}
  if(accessLost||version!==fileSelectionVersion)return;
  selectedFiles.push(...added);fileSelectionVersion++;$('work-files').value='';renderFiles();changed();
