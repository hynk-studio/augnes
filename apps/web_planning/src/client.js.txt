async function request(path,body){
  if(accessLost)throw new Error('access_denied');
  const response=await fetch(path,{method:body===undefined?'GET':'POST',credentials:'same-origin',cache:'no-store',redirect:'manual',
    headers:{'X-Requested-With':'XMLHttpRequest',...(body===undefined?{}:{'Content-Type':'application/json','X-CSRF-Token':csrf})},body:body===undefined?undefined:JSON.stringify({...scope,...body}),signal:AbortSignal.timeout(15000)});
  // Access can return 401/403, a login redirect, or an HTML challenge. None is
  // product data. An ordinary storage/transport failure still retains the draft.
  if(accessLost||response.status===403||response.status===401||response.type==='opaqueredirect'||
    (response.ok&&response.headers.get('X-Web-Planning-Response')!=='1')){deny();throw new Error('access_denied');}
  if(!response.ok){let code='storage_unavailable';try{code=(await response.json()).error;}catch{};const e=new Error(code);e.status=response.status;throw e;}
  return response;
}

function validCapacity(data,check,expected){
  const q=value=>value && Number.isSafeInteger(value.limit) && value.limit>0 && (value.used===null?value.remaining===null:Number.isSafeInteger(value.used)&&value.used>=0&&value.remaining===value.limit-value.used);
  return data?.format==='web_planning_capacity.v0.2' && data.check_id===check && JSON.stringify(data.expected)===JSON.stringify(expected) &&
    Number.isSafeInteger(data.history?.revisions) && data.history.revisions>=0 && Number.isSafeInteger(data.history.read_row_limit) && Number.isSafeInteger(data.history.read_byte_limit) && typeof data.fits==='boolean' && typeof data.noop==='boolean' && Array.isArray(data.issues) && data.issues.every(i=>typeof i.code==='string' && /^[a-z_]+$/.test(i.code) && (i.note===null||Number.isSafeInteger(i.note))) &&
    ['notes','selection','definition','relations'].every(key=>q(data[key])) && Array.isArray(data.rows) && data.rows.length===(data.notes.used!==null&&data.notes.used<=8?data.notes.used:0) &&
    data.rows.every(r=>['characters','source_units','bytes','text_bytes','duplicate_of'].every(k=>r[k]===null||(Number.isSafeInteger(r[k])&&r[k]>=0))) &&
    Number.isSafeInteger(data.note_character_limit) && Number.isSafeInteger(data.source_unit_limit) && Number.isSafeInteger(data.request_byte_limit) &&
    (data.selection.used===null || (Number.isSafeInteger(data.selection.text_bytes)&&data.selection.text_bytes>=0&&data.selection.metadata_bytes>=0&&data.selection.text_bytes+data.selection.metadata_bytes===data.selection.used)) &&
    (!data.fits || (!data.issues.length && ['notes','selection','definition','relations'].every(k=>data[k].used!==null&&data[k].remaining>=0) && data.rows.every(r=>r.characters!==null&&r.bytes!==null&&r.text_bytes!==null&&r.source_units!==null)));
}
